## Hướng Dẫn Sử Dụng Trong Next.js

```tsx
import Image from "next/image";
import { getEventVisual, getAllEvents, getEventsByCategory } from "@/lib/event-visuals";

// 1. Lấy thông tin một sự kiện cụ thể
const event = getEventVisual("hoi-nghi-khoa-hoc-y-te");

// 2. Lấy danh sách sự kiện theo chuyên mục
const conferences = getEventsByCategory("conference");

// 3. Render trong component
export function EventCard() {
  if (!event) return null;
  return (
    <div className="event-card">
      <Image
        src={event.imageSrc}
        alt={event.imageAlt}
        width={1280}
        height={720}
        className="rounded-lg shadow-md object-cover"
      />
      <h3>{event.title}</h3>
      <p>{event.description}</p>
    </div>
  );
}
```
